document.querySelector('#start').onclick=async()=>{const result=document.querySelector('#result'),button=document.querySelector('#start');button.disabled=true;let left,right,context,source;try{context=new AudioContext();await context.resume();source=context.createOscillator();const destination=context.createMediaStreamDestination();source.frequency.value=440;source.connect(destination);source.start();left=new RTCPeerConnection({iceServers:[]});right=new RTCPeerConnection({iceServers:[]});left.onicecandidate=e=>{if(e.candidate)right.addIceCandidate(e.candidate).catch(()=>{});};right.onicecandidate=e=>{if(e.candidate)left.addIceCandidate(e.candidate).catch(()=>{});};destination.stream.getTracks().forEach(t=>left.addTrack(t,destination.stream));await left.setLocalDescription(await left.createOffer());await right.setRemoteDescription(left.localDescription);await right.setLocalDescription(await right.createAnswer());await left.setRemoteDescription(right.localDescription);result.textContent='Đang truyền audio tổng hợp…';const started=Date.now();while(Date.now()-started<8000){await new Promise(r=>setTimeout(r,500));let bytes=0;for(const stat of (await right.getStats()).values())if(stat.type==='inbound-rtp'&&stat.kind==='audio')bytes+=stat.bytesReceived||0;if(bytes>0){result.textContent=JSON.stringify({passed:true,realRtpAudioBytesReceived:bytes,peerConnection:right.connectionState,microphoneUsed:false});return;}}throw Error('Không nhận được RTP audio trong 8 giây.');}catch(e){result.textContent=JSON.stringify({passed:false,error:e.message});}finally{source?.stop();left?.close();right?.close();await context?.close();button.disabled=false;}};
